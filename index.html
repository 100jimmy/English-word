// 升級版：新增單字後彈出成功提示視窗 async function addWord(e) { e.preventDefault(); const spelling = document.getElementById("word-spelling").value.trim(); const sentence = document.getElementById("word-sentence").value.trim(); const meaningsList = getMeaningsFromContainer("add-meanings-container"); if (meaningsList.length === 0) return alert("請至少填寫一組詞性與中文解釋！"); const primaryPos = meaningsList[0].pos; const primaryMeaning = meaningsList.map(m => `${m.pos} ${m.meaning}`).join(' '); vocabData.push({ id: Date.now().toString(), word: spelling, pos: primaryPos, meaning: primaryMeaning, meanings: meaningsList, sentence: sentence, isError: false }); renderWordList(); await saveUserDataToCloud(); // 清空表單欄位 document.getElementById("add-word-form").reset(); document.getElementById("add-meanings-container").innerHTML = ""; addMeaningRow("add-meanings-container"); if(document.getElementById("auto-fetch-msg")) document.getElementById("auto-fetch-msg").innerText = ""; // 彈出成功提示小視窗 showSuccessModal(spelling); } // 建立並顯示「加入成功」提示視窗的函式 function showSuccessModal(word) { // 檢查是否已有提示視窗，沒有則動態建立 let successModal = document.getElementById("success-modal"); if (!successModal) { successModal = document.createElement("div"); successModal.id = "success-modal"; successModal.className = "modal-overlay"; document.body.appendChild(successModal); } successModal.innerHTML = `
🎉

新增成功！

單字 ${word} 已成功同步至雲端單字庫。

`; successModal.classList.remove("hidden"); } // 關閉成功提示視窗並切換分頁 function closeSuccessModal(goToOverview = false) { const successModal = document.getElementById("success-modal"); if (successModal) { successModal.classList.add("hidden"); } if (goToOverview) { switchTab('list'); } else { // 聚焦回輸入框方便連續輸入 const spellingInput = document.getElementById("word-spelling"); if (spellingInput) spellingInput.focus(); } }
